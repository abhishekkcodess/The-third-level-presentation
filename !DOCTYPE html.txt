            document.querySelectorAll('.slide').forEach(s => s.classList.remove('active'));
            if (n > totalSlides) currentSlide = totalSlides;
            if (n < 1) currentSlide = 1;
            document.getElementById(`slide-${currentSlide}`).classList.add('active');
            document.getElementById('counter').innerText = `Slide ${currentSlide} of ${totalSlides}`;
        }

        function nextSlide() { currentSlide++; showSlide(currentSlide); }
        function prevSlide() { currentSlide--; showSlide(currentSlide); }

        // Keyboard Navigation
        document.addEventListener('keydown', (e) => {
            if (e.key === 'ArrowRight' || e.key === 'PageDown') nextSlide();
            if (e.key === 'ArrowLeft' || e.key === 'PageUp') prevSlide();
        });

        // Quiz Interactive Check
        function checkAnswer(btn, isCorrect) {
            const parent = btn.parentElement;
            parent.querySelectorAll('.quiz-btn').forEach(b => {
                b.style.pointerEvents = 'none';
                if(b.textContent.includes('B)')) {
                    b.classList.add('correct');
                }
            });
            if(!isCorrect) {
                btn.classList.add('incorrect');
            }
        }
    </script>
</body>
